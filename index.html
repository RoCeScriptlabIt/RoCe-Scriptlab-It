async function submitForm(event) { event.preventDefault(); hideFeedback(); const dict = i18n[currentLang] || i18n.it; const messaggio = document.getElementById("messaggio").value.trim(); if (!WEB_APP_URL || !WEB_APP_URL.startsWith("https://script.google.com/macros/s/")) { showFeedback("err", dict.msgMissingUrl); return; } const btn = document.getElementById("submitBtn"); btn.disabled = true; btn.textContent = dict.btnSending; const payload = { azione: "RICHIESTA_WEB", tipo_richiesta: currentTipo, lingua: currentLang, oggetto_contatto: document.getElementById("oggetto_contatto").value.trim(), chiave_licenza: document.getElementById("chiave_licenza").value.trim(), ambito: document.getElementById("ambito").value.trim(), tipo_diritti: document.getElementById("tipo_diritti").value, script_richiesto: document.getElementById("script_richiesto").value.trim(), num_postazioni: document.getElementById("num_postazioni").value, tipo_soggetto: document.getElementById("tipo_soggetto").value, nome: document.getElementById("nome").value.trim(), email: document.getElementById("email").value.trim(), os: document.getElementById("os").value.trim(), messaggio: messaggio }; try { await fetch(WEB_APP_URL, { method: "POST", mode: "no-cors", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(payload) }); showFeedback("ok", dict.msgSentOk); document.getElementById("roceForm").reset(); document.getElementById("tipo_soggetto").value = "Azienda"; } catch (err) { showFeedback("err", dict.msgSentErr + err.message); } finally { btn.disabled = false; btn.textContent = dict.btnSubmit; } }
